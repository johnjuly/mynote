    class Fraction//分数可以转换为小数
{
  public:
    Fraction(int num,int den=1)
    ：m_numerator(nums),m_denominatorm_numerator(den){}
    operator double() const{
      //没有返回类型 不改变 数据 为cosnt

      int m_numerator;
